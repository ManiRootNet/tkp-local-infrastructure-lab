; ----------------------------------------------------------
; Forward Lookup Zone: TKP.local
; Primary: SRV-1 (10.10.10.1)  |  Secondary: SRV-2, SRV-01
; DNSSEC: Signed
; ----------------------------------------------------------

; A records
www              IN  A       10.10.10.2      ; IIS web server (SRV-2)

; CNAME records
ftp              IN  CNAME   www.TKP.local.  ; FTP service alias -> SRV-2
dfs              IN  CNAME   www.TKP.local.  ; DFS namespace alias -> SRV-2

; MX record
@                IN  MX  10  srv-2.TKP.local.
; Note: no real mail server is running on SRV-2 — this record
; exists to demonstrate correct MX configuration only.

; NS records
@                IN  NS      srv-1.TKP.local.
@                IN  NS      srv-2.TKP.local.
@                IN  NS      srv-01.TKP.local.


; ----------------------------------------------------------
; Reverse Lookup Zone: 10.10.10.in-addr.arpa  (LAN A)
; Primary: SRV-1  |  Secondary: SRV-2, SRV-01
; DNSSEC: Signed
; ----------------------------------------------------------

1                IN  PTR     srv-1.TKP.local.
2                IN  PTR     srv-2.TKP.local.
@                IN  NS      srv-1.TKP.local.


; ----------------------------------------------------------
; Reverse Lookup Zone: 20.20.20.in-addr.arpa  (LAN B)

; DNSSEC: Signed
; ----------------------------------------------------------

1                IN  PTR     srv-3.TKP.local.
@                IN  NS      srv-1.TKP.local.


; ----------------------------------------------------------
; Reverse Lookup Zone: 60.60.60.in-addr.arpa  (LAN F)

; ----------------------------------------------------------

1                IN  PTR     srv-02.TKP.local.
